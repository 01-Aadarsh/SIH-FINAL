# Text-to-speech — Sarvam AI's Bulbul model (api/tts.py), opt-in via
# QueryRequest.synthesize_audio. Reuses SARVAM_API_KEY above — no separate
# key or manual model-terms approval needed (this replaced an earlier
# Groq-based implementation for exactly that reason: Groq's TTS model
# required an org admin to manually accept its usage terms in the Groq
# console before any request would work at all, and even then was
# English-only). Bulbul supports 11 of the languages SARVAM_API_KEY can
# translate into (see api/tts.py::BULBUL_SUPPORTED_LANGUAGES) and speaks
# the answer in that language directly; anything else skips audio rather
# than guessing. SARVAM_TTS_MODEL/SARVAM_TTS_SPEAKER have working defaults
# (bulbul:v3 / shubh, Sarvam's own documented default speaker) — only set
# these if you want a different one of the 36 available voices.

SARVAM_TTS_SPEAKER=shubh

# leaves margin. Longer answers are split at sentence boundaries into
